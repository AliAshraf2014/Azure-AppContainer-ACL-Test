#!/usr/bin/env bash
# Deploy datalake-acl-test to Container Apps
# Reads ACL from read-only Data Lake via DefaultAzureCredential / MI.
#
# Run from repo root:
#   bash steps.txt
#
# Safe to paste too: whole deploy runs in a subshell, so a failure
# will NOT close your Git Bash / Cloud Shell terminal.
# Target: s3-cnapp-intellixcore3-test (UAMI must already be assigned)
(
set -euo pipefail

# Strip Windows CR from az -o tsv output (Git Bash)
tsv() { tr -d '\r'; }

az extension add --name containerapp --upgrade 2>/dev/null || true

if ! az account show >/dev/null 2>&1; then
  az login
fi
az account set --subscription "b68899fa-2ea6-4081-b7aa-41e8cbb6f9ad"

RESOURCE_GROUP="s3-rg-intellixcore-uksouth-0001"
LOCATION="uksouth"
ACR_NAME="s3acrintellixcoreuksouth0001"
ACR_LOGIN_SERVER="${ACR_NAME}.azurecr.io"
IMAGE_REPO="blob-mi-test"
CONTAINER_APP_NAME="s3-cnapp-intellixcore3-test"
CONTAINER_PORT="8080"

IMAGE_TAG="v$(date -u +%Y%m%d%H%M%S)"
REVISION_SUFFIX="d$(date -u +%Y%m%d%H%M%S)"
FULL_IMAGE="${ACR_LOGIN_SERVER}/${IMAGE_REPO}:${IMAGE_TAG}"

echo "IMAGE_TAG=$IMAGE_TAG"
echo "REVISION_SUFFIX=$REVISION_SUFFIX"
echo "FULL_IMAGE=$FULL_IMAGE"

az acr build \
  --registry "$ACR_NAME" \
  --image "${IMAGE_REPO}:${IMAGE_TAG}" \
  --image "${IMAGE_REPO}:latest" \
  --file Dockerfile \
  .

# Resolve digest from the tag we just built (do NOT use show-manifests).
IMAGE_DIGEST=$(az acr repository show \
  --name "$ACR_NAME" \
  --image "${IMAGE_REPO}:${IMAGE_TAG}" \
  --query "digest" -o tsv | tsv)
if [[ -z "$IMAGE_DIGEST" ]]; then
  echo "ERROR: Image ${IMAGE_REPO}:${IMAGE_TAG} not found in ACR after build."
  exit 1
fi
FULL_IMAGE="${ACR_LOGIN_SERVER}/${IMAGE_REPO}@${IMAGE_DIGEST}"
echo "Deploying ${IMAGE_REPO}:${IMAGE_TAG} (${IMAGE_DIGEST})"

AZURE_CLIENT_ID=$(az containerapp show \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --query "identity.userAssignedIdentities | values(@)[0].clientId" -o tsv | tsv)
if [[ -z "$AZURE_CLIENT_ID" ]]; then
  echo "ERROR: No user-assigned identity on $CONTAINER_APP_NAME. Assign it in portal Identity first."
  exit 1
fi
echo "AZURE_CLIENT_ID=$AZURE_CLIENT_ID"

# --- Registry + ingress (application-scope) ---
az acr update --name "$ACR_NAME" --admin-enabled true

ACR_USERNAME=$(az acr credential show --name "$ACR_NAME" --query username -o tsv | tsv)
ACR_PASSWORD=$(az acr credential show --name "$ACR_NAME" --query "passwords[0].value" -o tsv | tsv)
az containerapp registry set \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --server "$ACR_LOGIN_SERVER" \
  --username "$ACR_USERNAME" \
  --password "$ACR_PASSWORD"

# Enable ingress if missing; otherwise just set target port.
if az containerapp ingress show \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" >/dev/null 2>&1; then
  az containerapp ingress update \
    --name "$CONTAINER_APP_NAME" \
    --resource-group "$RESOURCE_GROUP" \
    --target-port "$CONTAINER_PORT"
else
  az containerapp ingress enable \
    --name "$CONTAINER_APP_NAME" \
    --resource-group "$RESOURCE_GROUP" \
    --type external \
    --target-port "$CONTAINER_PORT" \
    --transport auto
fi

# --- Single revision-scope update: image + env + forced new revision ---
az containerapp update \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --image "$FULL_IMAGE" \
  --revision-suffix "$REVISION_SUFFIX" \
  --set-env-vars \
    "PORT=${CONTAINER_PORT}" \
    "LISTEN_HOST=0.0.0.0" \
    "AZURE_CLIENT_ID=${AZURE_CLIENT_ID}" \
    "DEPLOY_TAG=${IMAGE_TAG}"

echo "Waiting for new revision to provision..."
sleep 15

az containerapp revision list \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --query "[?properties.active].{name:name,replicas:properties.replicas,health:properties.healthState,image:properties.template.containers[0].image,suffix:properties.revisionSuffix}" \
  -o table

az containerapp show \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --query "{fqdn:properties.configuration.ingress.fqdn,latestRevision:properties.latestRevisionName,azureClientId:properties.template.containers[0].env[?name=='AZURE_CLIENT_ID'].value | [0],deployTag:properties.template.containers[0].env[?name=='DEPLOY_TAG'].value | [0]}" \
  -o json

FQDN=$(az containerapp show \
  --name "$CONTAINER_APP_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --query "properties.configuration.ingress.fqdn" -o tsv | tsv)

if [[ -n "$FQDN" ]]; then
  echo "Health: https://${FQDN}/health"
  curl -fsS "https://${FQDN}/health" || echo "WARN: health check failed (revision may still be starting)"
  echo
  echo "ACL smoke: POST https://${FQDN}/smoke"
  curl -fsS -X POST "https://${FQDN}/smoke" || echo "WARN: /smoke failed — check MI Data Lake permissions"
  echo
fi

echo
echo "=== Deploy finished successfully ==="
echo "Interactive exec (run separately):"
echo "  LATEST_REVISION=\$(az containerapp show -n $CONTAINER_APP_NAME -g $RESOURCE_GROUP --query properties.latestRevisionName -o tsv | tr -d '\\r')"
echo "  az containerapp exec -n $CONTAINER_APP_NAME -g $RESOURCE_GROUP --revision \"\$LATEST_REVISION\" --command /bin/sh"
echo "  # inside container: python test_azure_blob_mi.py"
)

echo "Script subshell exited with code $? (terminal stays open)."
